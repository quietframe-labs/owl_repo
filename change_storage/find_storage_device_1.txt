# Step 2. Stop pi-security-recorder.service.
## Stop recording:
    sudo systemctl stop pi-security-recorder.service
## Verify daemon stopped:
    systemctl is-active pi-security-recorder.service
## Verify drive isn't mounted:
    findmnt /dev/sda1
## Output:
    Nothing

# Step 4. Format it ext4.
## Format drive
    sudo mkfs.ext4 -L PI_SECURITY_NEW /dev/sda1
## Verify:
    lsblk -o NAME,SIZE,FSTYPE,LABEL,UUID,MOUNTPOINTS,MODEL

# Step 5. Update /etc/fstab with the new UUID..
## Update /etc/fstab 
    sudo nano /etc/fstab
### Old:
    UUID=c2a58269-5ea0-4844-bb98-313fd72e1a6d /srv/pi-security/storage ext4 defaults,nofail 0 2
### New:
    UUID=6f03038b-7e70-4431-8dbb-33f4a1e66b2c /srv/pi-security/storage ext4 defaults,nofail 0 2
# Step 6. Mount it.
## Mount drive:
    sudo systemctl daemon-reload
    sudo mount -a
## Check:
    findmnt /srv/pi-security/storage    
    df -h /srv/pi-security/storage
### Output:
    admin@owl:~$ findmnt /srv/pi-security/storage
    TARGET                   SOURCE    FSTYPE OPTIONS
    /srv/pi-security/storage /dev/sda1 ext4   rw,relatime
    admin@owl:~$ df -h /srv/pi-security/storage
    Filesystem      Size  Used Avail Use% Mounted on
    /dev/sda1       916G  2.1M  870G   1% /srv/pi-security/storage
    admin@owl:~$ 
# Step 7. Re-Create storage directories or copy old.
## On pi:
    sudo mkdir -p /srv/pi-security/storage/recordings/camera01
    sudo mkdir -p /srv/pi-security/storage/events
    sudo mkdir -p /srv/pi-security/storage/snapshots
## Set ownership:
    sudo chown -R admin:admin /srv/pi-security/storage
## Verify:
    ls -lah /srv/pi-security/storage

# Step 11. Restart the recorder and retention system.
## Start recorder:
    sudo systemctl start pi-security-recorder.service
## Verify:
    systemctl status pi-security-recorder.service --no-pager
    systemctl is-active pi-security-recorder.service

# Step 12. Confirm new recordings are actually landing on the 1 TB SSD.
## Wait 30-60 seconds:
    ls -lh /srv/pi-security/storage/recordings/camera01 | tail
    df -h /srv/pi-security/storage
