# Step 1: 
### Add venv to project:
#### from project home
    python -m venv .venv
### Activate venv
#### for powershell
    .\.venv\Scripts\Activate.ps1

# Step 2: Install django
    pip install django a2wsgi
#### FastAPI recommends the separate a2wsgi package when using django
### then:
    pip freeze > requirments.txt

# Step 3: Create django project:
### move into web
    cd .\web\
### start django
    django-admin startproject dashboard
#### See Django Quickstart.md

    └── everything else          Django

# Step 5: Install the integration depency
    python -m pip install a2wsgi
    python -m django --version
    python -m pip show a2wsgi
### Update requirements:
    python -m pip freeze > requirements.txt

    dashboard_app/
    ├── templates/

            ├── css/
            │   └── app.css
            └── js/
                └── hls.min.js
### Inside dashboard_app:
    mkdir static\dashboard_app\css -Force
    mkdir static\dashboard_app\js -Force
# Step 7: Move hls.min.js into Django static files
#### hls.min.js lives on the pi
### On pi run:
    sudo find /srv/pi-security -type f -name "hls.min.js" 2>/dev/null
### On PC run: 
    scp admin@owl:/srv/pi-security/web/static/js/hls.min.js .
### Move into Django static folder:
    Move-Item .\hls.min.js .\dashboard_app\static\dashboard_app\js\hls.min.js
### Confirm move:
    dir .\dashboard_app\static\dashboard_app\js

# Step 8: Update dashboard/settings.py
### Open settings.py
#### see note "step8"

# Step 9: Update dashboard_app/urls.py

# Step 10: Move django to pi
### On pi:
    systemctl show pi-security-web \
        -p WorkingDirectory \
        -p ExecStart
    sudo systemctl cat pi-security-web
#### Move main.py to PC
### on PC:
    scp admin@owl:/srv/pi-security/web/main.py .\web\main.py




### Reload and restart:
    sudo systemctl daemon-reload
    sudo systemctl restart pi-security-web
    sleep 2
