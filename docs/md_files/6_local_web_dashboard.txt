        app = FastAPI(
        title="Pi Security API",

        )

        STORAGE_PATH = Path("/srv/pi-security/storage")
        RECORDINGS_PATH = STORAGE_PATH / "recordings" / "camera01"


        def service_active(service_name: str) -> bool:
        result = subprocess.run(
                ["systemctl", "is-active", service_name],
                capture_output=True,
                text=True
        )

        return result.stdout.strip() == "active"


        @app.get("/")
        def root():
        return {
                "name": "Pi Security",
                "status": "online"
        }


        @app.get("/api/status")
        def system_status():
        usage = shutil.disk_usage(STORAGE_PATH)

        latest_recording = None

        if RECORDINGS_PATH.exists():
                recordings = list(RECORDINGS_PATH.glob("*.mp4"))

                if recordings:
                newest = max(recordings, key=lambda file: file.stat().st_mtime)

                latest_recording = {
                        "filename": newest.name,
                        "size_bytes": newest.stat().st_size,
                        "modified": newest.stat().st_mtime
                }

        return {
                "storage": {
                "mounted": STORAGE_PATH.is_mount(),
                "total_bytes": usage.total,
                "used_bytes": usage.used,
                "free_bytes": usage.free,
                "percent_used": round((usage.used / usage.total) * 100, 2)
                },

                "services": {
                "recorder": service_active("pi-security-recorder.service"),
                "retention_timer": service_active("pi-security-retention.timer"),
                "storage_monitor_timer": service_active(
                        "pi-security-storage-monitor.timer"
                )
                },

                "latest_recording": latest_recording
        }

        function setServiceStatus(elementId, active) {
        const element = document.getElementById(elementId);

        if (active) {
                element.textContent = "ACTIVE";
                element.className = "value online";
        } else {
                element.textContent = "OFFLINE";
                element.className = "value offline";
        }
        }


        function formatBytes(bytes) {
        const gb = bytes / (1024 ** 3);

        if (gb >= 1) {
                return gb.toFixed(1) + " GB";
        }

        const mb = bytes / (1024 ** 2);
        return mb.toFixed(1) + " MB";
        }


        async function updateDashboard() {

        try {

                const response = await fetch("/api/status");

                if (!response.ok) {
                throw new Error("API request failed");
                }

                const data = await response.json();

                setServiceStatus(
                "recorder",
                data.services.recorder
                );

                setServiceStatus(
                "retention",
                data.services.retention_timer
                );

                setServiceStatus(
                "monitor",
                data.services.storage_monitor_timer
                );


                const percent = data.storage.percent_used;

                document.getElementById("storage").textContent =
                percent.toFixed(2) + "% used";

                document.getElementById("storage-fill").style.width =
                Math.max(percent, 1) + "%";


                const latest = data.latest_recording;

                if (latest) {

                document.getElementById("latest-recording").innerHTML =
                        "<strong>" + latest.filename + "</strong><br>" +
                        formatBytes(latest.size_bytes);

                } else {

                document.getElementById("latest-recording").textContent =
                        "No recordings found";
                }


                const healthy =
                data.storage.mounted &&
                data.services.recorder &&
                data.services.retention_timer &&
                data.services.storage_monitor_timer;


                const overall = document.getElementById("overall-status");

                if (healthy) {
                overall.textContent = "● SYSTEM OK";
                overall.className = "online";
                } else {
                overall.textContent = "● SYSTEM WARNING";
                overall.className = "offline";
                }

        }

        catch (error) {

                const overall = document.getElementById("overall-status");

                overall.textContent = "● API OFFLINE";
                overall.className = "offline";

                console.error(error);
        }
        }


        updateDashboard();

        setInterval(updateDashboard, 5000);

        @app.get("/recordings/{filename}")
        def get_recording(filename: str):
        recording = (RECORDINGS_PATH / filename).resolve()

        if recording.parent != RECORDINGS_PATH.resolve():
                raise HTTPException(
                status_code=400,
                detail="Invalid filename"
                )

        if not recording.exists():
                raise HTTPException(
                status_code=404,
                detail="Recording not found"
                )

        if recording.suffix.lower() != ".mp4":
                raise HTTPException(
                status_code=400,
                detail="Invalid recording type"
                )

        return FileResponse(
                recording,
                media_type="video/mp4",
                filename=recording.name
        )

        @app.get("/api/recordings")
        def list_recordings():
        if not RECORDINGS_PATH.exists():
                return []

        recordings = []

        for recording in RECORDINGS_PATH.glob("*.mp4"):
                stat = recording.stat()

                recordings.append({
                "filename": recording.name,
                "size_bytes": stat.st_size,
                "modified": stat.st_mtime
                })

        recordings.sort(
                key=lambda item: item["modified"],
                reverse=True
        )

        return recordings

        function formatBytes(bytes) {
        const mb = bytes / (1024 ** 2);

        if (mb >= 1024) {
                return (mb / 1024).toFixed(2) + " GB";
        }

        return mb.toFixed(1) + " MB";
        }


        function formatRecordingTime(filename) {

        const match = filename.match(
                /(\d{4})-(\d{2})-(\d{2})_(\d{2})-(\d{2})-(\d{2})/
        );

        if (!match) {
                return filename;
        }

        const [
                ,
                year,
                month,
                day,
                hour,
                minute,
                second
        ] = match;

        return `${month}/${day}/${year} ${hour}:${minute}:${second}`;
        }


        function playRecording(filename) {

        const player =
                document.getElementById("video-player");

        const container =
                document.getElementById("video-container");

        document.getElementById("now-playing").textContent =
                "Now Playing: " + filename;

        player.src =
                "/recordings/" + encodeURIComponent(filename);

        container.style.display = "block";

        player.load();
        player.play();

        container.scrollIntoView({
                behavior: "smooth"
        });
        }


        async function loadRecordings() {

        const container =
                document.getElementById("recordings-list");

        try {

                const response =
                await fetch("/api/recordings");

                if (!response.ok) {
                throw new Error("Failed to load recordings");
                }

                const recordings =
                await response.json();

                if (recordings.length === 0) {
                container.textContent =
                        "No recordings found.";
                return;
                }

                container.innerHTML = "";

                recordings.forEach(recording => {

                const row =
                        document.createElement("div");

                row.className = "recording";

                const info =
                        document.createElement("div");

                info.className =
                        "recording-info";

                const filename =
                        document.createElement("div");

                filename.className =
                        "filename";

                filename.textContent =
                        recording.filename;

                const details =
                        document.createElement("div");

                details.className =
                        "details";

                details.textContent =
                        formatRecordingTime(
                        recording.filename
                        ) +
                        " • " +
                        formatBytes(
                        recording.size_bytes
                        );

                const button =
                        document.createElement("button");

                button.textContent =
                        "Play";

                button.addEventListener(
                        "click",
                        () =>
                        playRecording(
                                recording.filename
                        )
                );

                info.appendChild(filename);
                info.appendChild(details);

                row.appendChild(info);
                row.appendChild(button);

                container.appendChild(row);
                });

        }

        catch (error) {

                container.textContent =
                "Unable to load recordings.";

                console.error(error);
        }
        }


        loadRecordings();

        function formatBytes(bytes) {
        const mb = bytes / (1024 ** 2);

        if (mb >= 1024) {
                return (mb / 1024).toFixed(2) + " GB";
        }

        return mb.toFixed(1) + " MB";
        }


        function parseRecordingFilename(filename) {

        const match = filename.match(
                /(\d{4})-(\d{2})-(\d{2})_(\d{2})-(\d{2})-(\d{2})/
        );

        if (!match) {
                return null;
        }

        const [
                ,
                year,
                month,
                day,
                hour,
                minute,
                second
        ] = match;

        return new Date(
                Number(year),
                Number(month) - 1,
                Number(day),
                Number(hour),
                Number(minute),
                Number(second)
        );
        }


        function formatTime(date) {

        return date.toLocaleTimeString([], {
                hour: "numeric",
                minute: "2-digit",
                second: "2-digit"
        });
        }


        function formatDate(date) {

        return date.toLocaleDateString([], {
                weekday: "long",
                year: "numeric",
                month: "long",
                day: "numeric"
        });
        }


        function playRecording(filename) {

        const player =
                document.getElementById("video-player");

        const container =
                document.getElementById("video-container");

        document.getElementById("now-playing").textContent =
                "Now Playing: " + filename;

        player.src =
                "/recordings/" + encodeURIComponent(filename);

        container.style.display = "block";

        player.load();
        player.play();

        container.scrollIntoView({
                behavior: "smooth"
        });
        }


        async function loadRecordings() {

        const container =
                document.getElementById("recordings-list");

        const countElement =
                document.getElementById("recording-count");

        try {

                const response =
                await fetch("/api/recordings");

                if (!response.ok) {
                throw new Error("Failed to load recordings");
                }

                const recordings =
                await response.json();

                countElement.textContent =
                recordings.length === 1
                        ? "1 clip"
                        : recordings.length + " clips";

                if (recordings.length === 0) {

                container.innerHTML =
                        '<div class="empty-message">No recordings found.</div>';

                return;
                }

                container.innerHTML = "";

                const groups = {};

                recordings.forEach(recording => {

                const date =
                        parseRecordingFilename(recording.filename);

                if (!date) {
                        return;
                }

                const key =
                        date.toISOString().slice(0, 10);

                if (!groups[key]) {
                        groups[key] = [];
                }

                groups[key].push({
                        ...recording,
                        date: date
                });
                });


                Object.keys(groups)
                .sort()
                .reverse()
                .forEach(dateKey => {

                        const group =
                        document.createElement("section");

                        group.className =
                        "date-group";


                        const title =
                        document.createElement("h3");

                        title.className =
                        "date-title";

                        title.textContent =
                        formatDate(groups[dateKey][0].date);

                        group.appendChild(title);


                        groups[dateKey].forEach(recording => {

                        const row =
                                document.createElement("div");

                        row.className =
                                "recording";


                        const info =
                                document.createElement("div");

                        info.className =
                                "recording-info";


                        const time =
                                document.createElement("div");

                        time.className =
                                "recording-time";

                        time.textContent =
                                formatTime(recording.date);


                        const details =
                                document.createElement("div");

                        details.className =
                                "details";

                        details.textContent =
                                formatBytes(
                                recording.size_bytes
                                );


                        const filename =
                                document.createElement("div");

                        filename.className =
                                "filename";

                        filename.textContent =
                                recording.filename;


                        const button =
                                document.createElement("button");

                        button.textContent =
                                "Play";

                        button.addEventListener(
                                "click",
                                () =>
                                playRecording(
                                        recording.filename
                                )
                        );


                        info.appendChild(time);
                        info.appendChild(details);
                        info.appendChild(filename);

                        row.appendChild(info);
                        row.appendChild(button);

                        group.appendChild(row);
                        });


                        container.appendChild(group);
                });

        }

        catch (error) {

                countElement.textContent =
                "Unavailable";

                container.innerHTML =
                '<div class="empty-message">Unable to load recordings.</div>';

                console.error(error);
        }
        }


        loadRecordings();
