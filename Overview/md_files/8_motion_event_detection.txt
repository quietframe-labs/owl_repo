# Step 3: Create protected configuration file for camera
#### DO NOT hard-code camera password into py script
### Get working RTSP address:
    sudo systemctl cat pi-security-recorder.service 
### Then:
    sudo mkdir -p /etc/pi-security
    sudo nano /etc/pi-security/camera01-motion.env
#### paste:
    CAMERA_NAME=camera01

    RTSP_URL="rtsp://USERNAME:PASSWORD@CAMERA_IP:554/stream2"

    MOTION_WIDTH=320
    MOTION_HEIGHT=180
    MOTION_FPS=4

    PIXEL_DELTA=25
    MOTION_PERCENT=0.015

    START_FRAMES=3
    END_QUIET_SECONDS=6
    MIN_EVENT_SECONDS=2

    EVENT_RETENTION_DAYS=14
### Protect the file:
    sudo chown root:admin /etc/pi-security/camera01-motion.env
    sudo chmod 640 /etc/pi-security/camera01-motion.env
