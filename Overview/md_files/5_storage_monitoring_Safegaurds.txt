# Step 4: Add disk-usage monitoring
## 4.1 Create script:
    sudo nano /usr/local/bin/pi-security-storage-monitor.sh
### Paste:
-----------------------------------------------------------------------------
    #!/bin/bash

    MOUNT="/srv/pi-security/storage"
    LOG="/srv/pi-security/logs/storage-monitor.log"
    WARNING=80
    CRITICAL=90

    TIMESTAMP=$(date '+%Y-%m-%d %H:%M:%S')

    # Make sure the SSD is actually mounted
    if ! mountpoint -q "$MOUNT"; then
        echo "$TIMESTAMP ERROR: Storage SSD is not mounted at $MOUNT" >> "$LOG"
        exit 1
    fi

    # Get disk usage percentage
    USAGE=$(df --output=pcent "$MOUNT" | tail -1 | tr -dc '0-9')

    if [ "$USAGE" -ge "$CRITICAL" ]; then
        echo "$TIMESTAMP CRITICAL: Storage usage is ${USAGE}%" >> "$LOG"
        exit 2
    elif [ "$USAGE" -ge "$WARNING" ]; then
        echo "$TIMESTAMP WARNING: Storage usage is ${USAGE}%" >> "$LOG"
    else
        echo "$TIMESTAMP OK: Storage usage is ${USAGE}%" >> "$LOG"
    fi
-----------------------------------------------------------------------------

    # Step 5: Low-space protection
        0–79%   → Normal
        80–89%  → Warning
        90–94%  → Critical warning
        95%+    → Run retention cleanup immediately
        98%+    → Stop recorder to protect filesystem
    ## 5.1 Update storage monitor:
        sudo nano /usr/local/bin/pi-security-storage-monitor.sh
    ### Replace contents:
    #!/bin/bash

    MOUNT="/srv/pi-security/storage"
    LOG="/srv/pi-security/logs/storage-monitor.log"
    RETENTION_SCRIPT="/usr/local/bin/pi-security-retention.sh"

    WARNING=80
    CRITICAL=90
    CLEANUP=95
    EMERGENCY=98

    TIMESTAMP=$(date '+%Y-%m-%d %H:%M:%S')

    # Verify the SSD is mounted
    if ! mountpoint -q "$MOUNT"; then
        echo "$TIMESTAMP ERROR: Storage SSD is not mounted at $MOUNT" >> "$LOG"
        exit 1
    fi

    # Get disk usage percentage
    USAGE=$(df --output=pcent "$MOUNT" | tail -1 | tr -dc '0-9')

    # Emergency protection
    if [ "$USAGE" -ge "$EMERGENCY" ]; then
        echo "$TIMESTAMP EMERGENCY: Storage usage is ${USAGE}%. Stopping recorder." >> "$LOG"

        systemctl stop pi-security-recorder.service

        exit 3

    # Run retention early
    elif [ "$USAGE" -ge "$CLEANUP" ]; then
        echo "$TIMESTAMP CRITICAL: Storage usage is ${USAGE}%. Running retention cleanup." >> "$LOG"

        "$RETENTION_SCRIPT" >> "$LOG" 2>&1

        # Check usage again after cleanup
        NEW_USAGE=$(df --output=pcent "$MOUNT" | tail -1 | tr -dc '0-9')

        echo "$TIMESTAMP CLEANUP: Storage usage after cleanup is ${NEW_USAGE}%." >> "$LOG"

        exit 2

    elif [ "$USAGE" -ge "$CRITICAL" ]; then
        echo "$TIMESTAMP CRITICAL: Storage usage is ${USAGE}%" >> "$LOG"
        exit 2

    elif [ "$USAGE" -ge "$WARNING" ]; then
        echo "$TIMESTAMP WARNING: Storage usage is ${USAGE}%" >> "$LOG"

    else
        echo "$TIMESTAMP OK: Storage usage is ${USAGE}%" >> "$LOG"
    fi
-------------------------------------------------------------------------

# Step 6: Create Pi security health-check command
## Create script:
    sudo nano /usr/local/bin/pi-security-status.sh
## Paste:
----------------------------------------------------------------
    #!/bin/bash

    MOUNT="/srv/pi-security/storage"
    RECORDING_DIR="$MOUNT/recordings/camera01"

    echo "======================================"
    echo " Pi Security System Status"
    echo " $(date)"
    echo "======================================"
    echo

    echo "[Storage Mount]"
    if mountpoint -q "$MOUNT"; then
        echo "OK: SSD mounted at $MOUNT"
        findmnt "$MOUNT"
    else
        echo "ERROR: SSD is NOT mounted"
    fi

    echo
    echo "[Storage Usage]"
    df -h "$MOUNT" 2>/dev/null || echo "Unable to read storage usage"

    echo
    echo "[Recorder]"
    systemctl is-active --quiet pi-security-recorder.service \
        && echo "OK: Recorder is active" \
        || echo "ERROR: Recorder is not active"

    echo
    echo "[Retention Timer]"
    systemctl is-active --quiet pi-security-retention.timer \
        && echo "OK: Retention timer is active" \
        || echo "ERROR: Retention timer is not active"

    echo
    echo "[Storage Monitor Timer]"
    systemctl is-active --quiet pi-security-storage-monitor.timer \
        && echo "OK: Storage monitor timer is active" \
        || echo "ERROR: Storage monitor timer is not active"

    echo
    echo "[Latest Recording]"
    LATEST=$(find "$RECORDING_DIR" -type f -name "*.mp4" -printf '%T@ %p\n' 2>/dev/null \
        | sort -nr \
        | head -1 \
        | cut -d' ' -f2-)

    if [ -n "$LATEST" ]; then
        echo "Latest file:"
        ls -lh "$LATEST"
    else
        echo "ERROR: No recordings found"
    fi

    echo
    echo "[Recent Storage Monitor]"
    tail -3 /srv/pi-security/logs/storage-monitor.log 2>/dev/null \
        || echo "No storage-monitor log found"

    echo
    echo "======================================"
----------------------------------------------------------------
