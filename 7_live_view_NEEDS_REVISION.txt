# Step 2: Create the service
    sudo nano /etc/systemd/system/pi-security-live.service
#### Paste:
    [Unit]
    Description=Pi Security Live View Stream
    After=network-online.target
    Wants=network-online.target

    [Service]
    Type=simple
    User=admin

    ExecStartPre=/usr/bin/bash -c 'rm -f /srv/pi-security/live/camera01/*.m3u8 /srv/pi-security/live/camera01/*.ts'

    ExecStart=/usr/bin/ffmpeg \
        -rtsp_transport tcp \
        -i rtsp://CaptainOwl:N49VRJb2e2@192.168.86.237:554/stream1 \
        -map 0:v:0 \
        -c:v copy \
        -an \
        -f hls \
        -hls_time 2 \
        -hls_list_size 5 \
        -hls_flags delete_segments+append_list+omit_endlist \
        -hls_segment_filename /srv/pi-security/live/camera01/segment_%%05d.ts \
        /srv/pi-security/live/camera01/index.m3u8

    Restart=always
    RestartSec=5

    [Install]
    WantedBy=multi-user.target
