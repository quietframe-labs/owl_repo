## Recordings are going to ssd at:
    /srv/pi-security/storage/recordings/camera01/
## Keep 14 days of recordings and automatically delete anything older
    retention script
        ↓
    systemd service
        ↓
    systemd timer
        ↓
    runs automatically once per day
# Step 1: Create retention script
    sudo nano /usr/local/bin/pi-security-retention.sh
### Paste this:
    #!/bin/bash

    RECORDING_DIR="/srv/pi-security/storage/recordings"
    RETENTION_DAYS=14

    echo "Pi Security retention cleanup started: $(date)"

    find "$RECORDING_DIR" \                  
        -type f \
        -name "*.mp4" \
        -mtime +$RETENTION_DAYS \
        -print \
        -delete

    echo "Pi Security retention cleanup finished: $(date)"
## find "$RECORDING_DIR... Search recordings dir, only consider .mp4, older than 14 days, print in log and delete
## Make it executable:
    sudo chmod +x /usr/local/bin/pi-security-retention.sh    

# Step 2: Test
## Before automatic deletion:
    sudo /usr/local/bin/pi-security-retention.sh
### Verify:
    ls -lh /srv/pi-security/storage/recordings/camera01 | tail

# Step 3: Automate it
## Create systemd service:
    sudo nano /etc/systemd/system/pi-security-retention.service
### Paste this:
    [Unit]
    Description=Pi Security Recording Retention Cleanup

    [Service]
    Type=oneshot
    ExecStart=/usr/local/bin/pi-security-retention.sh
## Create timer:
    sudo nano /etc/systemd/system/pi-security-retention.timer
### Paste this:
    [Unit]
    Description=Run Pi Security Retention Cleanup Daily

    [Timer]
    OnCalendar=daily
    Persistent=true

    [Install]
    WantedBy=timers.target
## Reload systemd and enable timer
    sudo systemctl daemon-reload
    sudo systemctl enable --now pi-security-retention.timer
### Verify:
    systemctl status pi-security-retention.timer
    systemctl list-timers --all | grep pi-security
## Useful command:
    journalctl -u pi-security-retention.service
### Lets you inspect the cleanup history later and see when the retention job ran and which files it deleted
